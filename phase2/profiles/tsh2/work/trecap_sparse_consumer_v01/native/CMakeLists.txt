cmake_minimum_required(VERSION 3.20)
project(trecap_sparse_benchmark LANGUAGES CXX)
add_executable(trecap_sparse_benchmark main.cpp codec.cpp consumer.cpp observer.cpp)
target_compile_features(trecap_sparse_benchmark PRIVATE cxx_std_20)
set_property(TARGET trecap_sparse_benchmark PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
if(MSVC)
  target_compile_options(trecap_sparse_benchmark PRIVATE /W4 /O2 /GL- /EHsc)
  target_link_options(trecap_sparse_benchmark PRIVATE /LTCG:OFF)
else()
  target_compile_options(trecap_sparse_benchmark PRIVATE -Wall -Wextra -Wpedantic -O3 -fno-lto)
  target_link_options(trecap_sparse_benchmark PRIVATE -fno-lto)
endif()
enable_testing()
add_test(NAME codec_contract COMMAND trecap_sparse_benchmark --self-test)
