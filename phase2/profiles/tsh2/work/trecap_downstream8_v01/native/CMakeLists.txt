cmake_minimum_required(VERSION 3.20)
project(trecap_downstream8 LANGUAGES CXX)
add_executable(trecap_downstream8 main.cpp codec.cpp consumer.cpp observer.cpp)
add_executable(codec_selftest codec_selftest_main.cpp codec.cpp consumer.cpp observer.cpp)
foreach(target trecap_downstream8 codec_selftest)
  target_compile_features(${target} PRIVATE cxx_std_20)
  set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /O2 /GL- /EHsc)
    target_link_options(${target} PRIVATE /LTCG:OFF)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -O3 -fno-lto)
    target_link_options(${target} PRIVATE -fno-lto)
  endif()
endforeach()
enable_testing()
add_test(NAME preserved_codec_contract COMMAND codec_selftest --self-test)
